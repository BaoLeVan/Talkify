// ── CONTACT / FRIEND TABLES (NEW) ──────────────────────────────────────────

table friendships {
  id bigint [primary key]           // Snowflake ID
  requester_id bigint [not null]    // FK → users.id — người gửi lời mời
  addressee_id bigint [not null]    // FK → users.id — người nhận lời mời
  status varchar [not null]         // PENDING | ACCEPTED | REJECTED | CANCELLED | REMOVED | BLOCKED
  blocked_by bigint                 // FK → users.id (nullable) — ai là người block
  created_at timestamp [not null]
  responded_at timestamp            // nullable — thời điểm chấp nhận/từ chối

  // UNIQUE CONSTRAINT: canonical pair (min(r,a), max(r,a)) — không cho phép duplicate
  // INDEX: (requester_id, status) — danh sách lời mời đã gửi
  // INDEX: (addressee_id, status) — danh sách lời mời nhận được
  // INDEX: (requester_id, addressee_id) UNIQUE — tìm nhanh quan hệ giữa 2 user
}
